@evaluate.run
def evaluate(data: dict):
    """
    Evaluate the feature transformations on wind power prediction task (ERA5 data).

    Workflow:
    1. Extract single grid template (13 columns: 7 weather + 6 time)
    2. Apply LLM's feature engineering to template
    3. Apply engineered template to all 4 grids
    4. Merge all grids + time features
    5. Train LGBM with K-Fold CV

    ERA5 特点：
    - 有100m高度数据（wind_speed_100m, wind_direction_100m）
    - 高质量再分析数据
    - 小时级时间分辨率
    """
    import sys
    from pathlib import Path
    import pandas as pd
    import numpy as np
    from sklearn.model_selection import KFold
    from sklearn.metrics import mean_squared_error

    # Add Wind project to Python path
    wind_project_path = '/data/cqj-project/wind-power-forecast-mlflow-1'
    if wind_project_path not in sys.path:
        sys.path.insert(0, wind_project_path)

    # Add LLM-FE utils to path
    llmfe_path = '/data/cqj-project/LLM-FE/LLMFE'
    if llmfe_path not in sys.path:
        sys.path.insert(0, llmfe_path)

    from core.models.base_lgbm import BaseLGBM
    from utils.grid_template import (
        extract_grid_template,
        apply_template_to_grid,
        merge_all_grids
    )

    # Load data
    template_inputs, outputs = data['inputs'], data['outputs']

    # ERA5 weather source detection (has wind_speed_100m)
    weather_source = 'era5'

    # Re-load the full multi-grid data from CSV
    import os
    data_dir = '/data/cqj-project/LLM-FE/LLMFE/data'
    data_path = os.path.join(data_dir, f'windpower_SD_W_B_{weather_source}.csv')

    if not os.path.exists(data_path):
        data_path = f'./data/windpower_SD_W_B_{weather_source}.csv'

    full_df = pd.read_csv(data_path)

    # Extract features and target
    target_col = full_df.columns[-1]
    full_inputs = full_df.drop(columns=[target_col])
    full_outputs = full_df[target_col]

    # Verify shape matches
    assert len(full_inputs) == len(template_inputs), f"Data length mismatch: {len(full_inputs)} vs {len(template_inputs)}"

    print(f"[DEBUG] template_inputs shape: {template_inputs.shape}")
    print(f"[DEBUG] template_inputs columns: {template_inputs.columns.tolist()}")

    # Apply LLM's feature engineering to template
    engineered_template = modify_features(template_inputs)
    print(f"[DEBUG] engineered_template shape: {engineered_template.shape}")
    print(f"[DEBUG] engineered_template columns: {engineered_template.columns.tolist()}")

    # Apply engineered template to all 4 grids
    all_grids_features = []

    for grid_id in [1, 2, 3, 4]:
        grid_features = apply_template_to_grid(
            template_df=engineered_template,
            original_inputs=full_inputs,
            grid_id=grid_id,
            modify_features_func=modify_features,
            source='era5'  # Explicitly specify ERA5
        )
        all_grids_features.append(grid_features)

    # Extract time features
    time_features = full_inputs[['hour_sin', 'hour_cos',
                                 'day_of_week_sin', 'day_of_week_cos',
                                 'month_sin', 'month_cos']]

    # Merge all grids + time features
    X = merge_all_grids(all_grids_features, time_features)

    # Convert y to pandas Series
    if isinstance(outputs, np.ndarray):
        y = pd.Series(outputs, index=X.index, name='power')
    else:
        y = outputs

    # Encode categorical columns if any
    for col in X.columns:
        if X[col].dtype == 'object' or X[col].dtype.name == 'string':
            X[col] = pd.Categorical(X[col]).codes

    # Handle missing values and infinite values
    X = X.replace([np.inf, -np.inf], np.nan)
    X = X.fillna(0)

    # Time-series K-Fold CV (no shuffle to preserve temporal order)
    kf = KFold(n_splits=4, shuffle=False)
    scores = []

    for train_idx, test_idx in kf.split(X, y):
        X_train, X_test = X.iloc[train_idx], X.iloc[test_idx]
        y_train, y_test = y.iloc[train_idx], y.iloc[test_idx]

        # Use Wind project's BaseLGBM
        model = BaseLGBM(
            model_name='llmfe_eval',
            weather_source='era5',
            params={
                'objective': 'regression',
                'metric': 'rmse',
                'learning_rate': 0.03,
                'num_leaves': 31,
                'max_depth': 8,
                'min_child_samples': 50,
                'feature_fraction': 0.9,
                'bagging_fraction': 0.8,
                'bagging_freq': 5,
                'reg_alpha': 0.1,
                'reg_lambda': 0.1,
                'verbose': -1
            }
        )

        try:
            model.train(
                X_train=X_train,
                y_train=y_train,
                X_val=X_test,
                y_val=y_test,
                num_boost_round=500,
                early_stopping_rounds=50
            )

            y_pred = model.predict(X_test)
            rmse = np.sqrt(mean_squared_error(y_test, y_pred))
            scores.append(-rmse)

        except Exception as e:
            print(f"Training error: {e}")
            scores.append(-999999)

    # Return average score
    avg_score = np.mean(scores)
    
    # 🆕 Generate CART feedback for LLM (OCTree-inspired)
    # IMPORTANT: Only generate CART if all folds succeeded (no -999999 scores)
    cart_feedback = None
    if all(score > -999999 for score in scores):  # All folds succeeded
        try:
            from llmfe.utils.cart_feedback import get_cart_feedback
            # Use the final X and y to train CART
            cart_feedback = get_cart_feedback(X, y, max_depth=3)
        except Exception as e:
            print(f"CART generation failed: {e}")
            cart_feedback = None
    else:
        print(f"Skip CART generation due to LGBM training failure")


    # Always return the ORIGINAL template
    original_template = extract_grid_template(full_inputs, grid_id=1, source='era5')
    return avg_score, original_template, outputs, cart_feedback


@equation.evolve
def modify_features(df_input) -> pd.DataFrame:
    """
    Feature engineering for wind power prediction using ERA5 reanalysis data.

    IMPORTANT: Input is from a SINGLE grid point with simplified column names:
    - relative_humidity_2m
    - wind_speed_10m
    - wind_direction_10m
    - wind_speed_100m         <- ERA5 unique: hub-height wind speed
    - wind_direction_100m     <- ERA5 unique: hub-height wind direction
    - pressure_msl
    - surface_pressure
    - hour_sin, hour_cos
    - day_of_week_sin, day_of_week_cos
    - month_sin, month_cos

    Your features will be automatically applied to all 4 grids.

    NOTE: This is TIME-SERIES data with hourly frequency. Consider exploring temporal patterns.
    NOTE: Apply principles from Wind Energy Handbook (atmospheric physics, turbulence, power curves) to create features.
    NOTE: ERA5 provides 100m height data - closer to typical wind turbine hub heights (80-120m).
    NOTE: Use 100m data for primary features, 10m for wind shear calculations.

    Thought 1: Wind power is proportional to wind speed cubed at hub height (100m is closer to hub than 10m).
    New Feature 1: wind_speed_100m_cubed | wind_speed_100m_cubed = (wind_speed_100m ** 3)
    """
    import pandas as pd
    import numpy as np

    df_output = df_input.copy()

    # Add cubic wind speed at hub height (100m)
    df_output['wind_speed_100m_cubed'] = df_input['wind_speed_100m'] ** 3

    return df_output
