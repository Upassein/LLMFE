@evaluate.run
def evaluate(data: dict):
    """
    Evaluate the feature transformations on wind power prediction task.

    New workflow:
    1. Extract single grid template (11 columns)
    2. Apply LLM's feature engineering to template
    3. Apply engineered template to all 4 grids
    4. Merge all grids + time features
    5. Train LGBM with K-Fold CV

    Uses Wind project's BaseLGBM with time-series K-Fold CV.
    """
    import sys
    from pathlib import Path
    import pandas as pd
    import numpy as np
    from sklearn.model_selection import KFold
    from sklearn.metrics import mean_squared_error

    # Add Wind project to Python path (modify this path on your server)
    wind_project_path = '/data/cqj-project/wind-power-forecast-mlflow-1'
    if wind_project_path not in sys.path:
        sys.path.insert(0, wind_project_path)

    # Add LLM-FE utils to path
    llmfe_path = '/data/cqj-project/LLM-FE/LLMFE'
    if llmfe_path not in sys.path:
        sys.path.insert(0, llmfe_path)

    from core.models.base_lgbm import BaseLGBM
    from utils.grid_template import (
        extract_grid_template,
        apply_template_to_grid,
        merge_all_grids
    )

    # Load data
    inputs, outputs = data['inputs'], data['outputs']

    # === Step 1: Extract grid1 as template (auto-detect source) ===
    template_df = extract_grid_template(inputs, grid_id=1, source=None)
    # template_df: N weather features + 6 time features
    # N depends on source: ecmwf(5), gfs(9), icon(11)

    # === Step 2: Apply LLM's feature engineering to template ===
    engineered_template = modify_features(template_df)
    # engineered_template: (N+6) + M columns (original + new features)

    # === Step 3: Apply engineered template to all 4 grids ===
    all_grids_features = []
    for grid_id in [1, 2, 3, 4]:
        grid_features = apply_template_to_grid(
            template_df=engineered_template,
            original_inputs=inputs,
            grid_id=grid_id,
            modify_features_func=modify_features,
            source=None  # Auto-detect
        )
        all_grids_features.append(grid_features)

    # === Step 4: Extract time features (global, not grid-specific) ===
    time_features = inputs[['hour_sin', 'hour_cos',
                            'day_of_week_sin', 'day_of_week_cos',
                            'month_sin', 'month_cos']]

    # === Step 5: Merge all grids + time features ===
    X = merge_all_grids(all_grids_features, time_features)

    # Convert y to pandas Series if it's numpy array (IMPORTANT for BaseLGBM compatibility)
    if isinstance(outputs, np.ndarray):
        y = pd.Series(outputs, index=X.index, name='power')
    else:
        y = outputs

    # Encode categorical columns if any
    for col in X.columns:
        if X[col].dtype == 'object' or X[col].dtype.name == 'string':
            X[col] = pd.Categorical(X[col]).codes

    # Handle missing values and infinite values
    X = X.replace([np.inf, -np.inf], np.nan)
    X = X.fillna(0)

    # Time-series K-Fold CV (no shuffle to preserve temporal order)
    kf = KFold(n_splits=4, shuffle=False)
    scores = []

    for train_idx, test_idx in kf.split(X, y):
        X_train, X_test = X.iloc[train_idx], X.iloc[test_idx]
        y_train, y_test = y.iloc[train_idx], y.iloc[test_idx]  # Use .iloc for Series

        # Use Wind project's BaseLGBM with faster training params
        model = BaseLGBM(
            model_name='llmfe_eval',
            weather_source='combined',
            params={
                'objective': 'regression',
                'metric': 'rmse',
                'learning_rate': 0.03,
                'num_leaves': 31,
                'max_depth': 8,
                'min_child_samples': 50,
                'feature_fraction': 0.9,
                'bagging_fraction': 0.8,
                'bagging_freq': 5,
                'reg_alpha': 0.1,
                'reg_lambda': 0.1,
                'verbose': -1
            }
        )

        try:
            # Train using Wind project's training method
            model.train(
                X_train=X_train,
                y_train=y_train,
                X_val=X_test,
                y_val=y_test,
                num_boost_round=500,
                early_stopping_rounds=50
            )

            # Predict and calculate RMSE
            y_pred = model.predict(X_test)
            rmse = np.sqrt(mean_squared_error(y_test, y_pred))
            scores.append(-rmse)  # Negative for maximization

        except Exception as e:
            # If training fails, return a very bad score
            print(f"Training error: {e}")
            scores.append(-999999)

    # Return average score
    avg_score = np.mean(scores)
    return avg_score, inputs, outputs


@equation.evolve
def modify_features(df_input) -> pd.DataFrame:
    """
    Feature engineering for wind power prediction.

    🔄 NEW WORKFLOW:
    You will receive data from a SINGLE meteorological grid point (not all 4 grids).
    Your feature engineering will be automatically applied to all 4 grids.
    Focus on creating physically meaningful features for one location.

    INPUT FEATURES (11 columns):

    Meteorological variables (5 columns):
    - relative_humidity_2m: Relative humidity at 2 meters (%)
    - wind_speed_10m: Wind speed at 10 meters height (m/s)
    - wind_direction_10m: Wind direction at 10 meters (degrees, 0-360)
    - pressure_msl: Mean sea level pressure (hPa)
    - surface_pressure: Surface pressure (hPa)

    Time features (6 columns):
    - hour_sin, hour_cos: Hour of day encoded as sin/cos (24-hour cycle)
    - day_of_week_sin, day_of_week_cos: Day of week encoded (7-day cycle)
    - month_sin, month_cos: Month of year encoded (12-month cycle)

    TASK: Predict wind power generation
    - Target range: 0-99 MW
    - Installed capacity: 99 MW
    - Mean power: ~20 MW

    CRITICAL RULES:
    1. ✅ Always start with: df_output = df_input.copy()
    2. ✅ Always return: return df_output
    3. ✅ DO NOT delete any original features - only ADD new features
    4. ✅ Create 5-10 new features per iteration
    5. ✅ Focus on simple, physically meaningful transformations

    PHYSICAL KNOWLEDGE (wind power fundamentals):

    1. Wind Power Law: Power ∝ wind_speed³
       → Most important feature: wind_speed^3

    2. Wind Direction (circular variable):
       → Convert to U/V components:
          U = speed × sin(direction_radians)
          V = speed × cos(direction_radians)

    3. Air Density affects power:
       → Higher pressure → higher density → more power
       → Higher humidity → lower density → less power
       → Density proxy: pressure_msl / surface_pressure

    4. Temporal patterns:
       → Wind speed varies by hour (diurnal cycle)
       → Seasonal effects in month features

    SUGGESTED FEATURES (start with simple ones):

    BASIC (try these first):
    ```python
    # Wind speed powers (MOST IMPORTANT)
    df_output['wind_speed_squared'] = df_input['wind_speed_10m'] ** 2
    df_output['wind_speed_cubed'] = df_input['wind_speed_10m'] ** 3

    # Wind direction decomposition (handle circular variable)
    wind_dir_rad = np.deg2rad(df_input['wind_direction_10m'])
    df_output['wind_u'] = df_input['wind_speed_10m'] * np.sin(wind_dir_rad)
    df_output['wind_v'] = df_input['wind_speed_10m'] * np.cos(wind_dir_rad)

    # Air density proxy
    df_output['pressure_ratio'] = df_input['pressure_msl'] / (df_input['surface_pressure'] + 1e-6)
    ```

    INTERACTIONS (try after basic features work):
    ```python
    # Speed × time (diurnal patterns)
    df_output['speed_hour_sin'] = df_input['wind_speed_10m'] * df_input['hour_sin']
    df_output['speed_hour_cos'] = df_input['wind_speed_10m'] * df_input['hour_cos']

    # Speed × humidity (air density effect)
    df_output['speed_humidity'] = df_input['wind_speed_10m'] * df_input['relative_humidity_2m']

    # Speed^3 × pressure (power with density correction)
    df_output['power_density_corrected'] = (df_input['wind_speed_10m'] ** 3) * df_input['pressure_msl']
    ```

    BINNING (categorize for non-linear patterns):
    ```python
    # Wind speed categories (cut-in, linear, rated, cut-out)
    df_output['speed_low'] = (df_input['wind_speed_10m'] < 3).astype(int)      # Cut-in
    df_output['speed_medium'] = ((df_input['wind_speed_10m'] >= 3) &
                                  (df_input['wind_speed_10m'] < 12)).astype(int)  # Linear
    df_output['speed_high'] = (df_input['wind_speed_10m'] >= 12).astype(int)   # Rated/Cut-out
    ```

    WHAT NOT TO DO:
    ❌ Don't create grid-specific features (e.g., grid1_wind_speed)
    ❌ Don't create spatial aggregations (mean/std across grids) - not visible to you
    ❌ Don't use complex nested transformations (log of log, etc.)
    ❌ Don't create more than 15 features at once

    BASELINE PERFORMANCE:

